% Milestone timeline for the historical development section.
% \input by chapters/02-literature-review.tex.
%
% LINEAR TIME AXIS: x = (year - 1932) * 0.155 cm. Deliberately no decade
% scale -- tick spacing itself carries elapsed time; every milestone prints
% its year.
% TWO TRACKS BY COLOUR, NOT BY SIDE OF THE AXIS: tlpositron = positron line,
% tlsingle = borrowed single-photon/transmission lineage. 1951, 1989 and 2013
% are positron entries sitting below purely for room -- do not "tidy" to grey.
%
% LAYOUT: the axis clusters too tightly for labels to sit over their ticks,
% so each entry carries year / row / label x, with a leader from tick to
% label. Rows step outward in 1.50 cm increments, two per side.
%
% CLEARANCES, VERIFIED BY MEASUREMENT (hook at the foot of the file) -- TeX
% is silent about overlapping labels and early attempts shipped with some:
%   Horizontal -- labels are 2.3 cm wide, so same-row label x must be
%   >= 2.6 cm apart. Tightest pairs: 1976/1986 and 1986/2006, 5.1 pt.
%   Vertical -- the 1.50 cm row gap must clear the TALLEST label (1953,
%   four lines, 47 pt); tightest pair 2006 vs row 2, 3.5 pt.
%   Leaders crossing labels is a third failure mode -- see elbow routing
%   below, which fixes label x for every row-2 entry.
% Total extent 15.7 cm, inside the 16 cm text block.

\providecolor{tlpositron}{HTML}{2A6F97}
\providecolor{tlsingle}{HTML}{8B9BA4}
\providecolor{tltext}{HTML}{3D4A52}

% LEADERS ARE ELBOWS, AND THE ROUTE IS LOAD-BEARING. A straight leader to
% row 2 cuts through the row-1 band and gets visibly broken by the labels'
% page-colour fill. Each leader instead runs up to y = 0.40 (below every
% label, which start at 0.55), across to its label x, then up -- with the
% vertical run threading a GAP between row-1 labels. Measured row-1 spans and
% the gaps threaded (pt):
%   above  1932 [-34,34]  1953 [57,125]  1976 [162,231]  1986 [236,305]  2006 [310,379]
%          1975 lx 153 (gap 125-162)   1983 lx 233 (gap 231-236)   2018 lx 381
%   below  1958 [77,145]  1973 [153,222]  1989 [230,299]  2013 [313,382]
%          1951 lx 65 (before 77)      1963 lx 150 (gap 145-153)
% Several clearances are only 2-4 pt: move ANY row-1 label and the row-2
% leader threading past it must move too. Labels draw on their own layer,
% filled with pagebg (must track the theme) as a backstop mask.
\pgfdeclarelayer{tllabels}
\pgfsetlayers{main,tllabels}

\begin{tikzpicture}[x=1cm, y=1cm, font=\scriptsize,
	tick/.style={line width=0.6pt},
	lead/.style={line width=0.3pt},
	lbl/.style={align=center, text width=2.3cm, text=tltext, inner sep=1.5pt,
		fill=pagebg}]

\draw[tltext, line width=0.8pt, -latex] (-0.45,0) -- (13.9,0);

% --- positron line (above) --  year/row/label x/label ----------------------
\foreach \year/\r/\lx/\text in {%
	1932/1/0/{Anderson\\positron discovered},
	1953/1/3.2/{Brownell \& Sweet\\rectilinear probes},
	1975/2/5.4/{PETT\\ring scanners},
	1976/1/6.9/{\textsuperscript{18}F-FDG\\synthesised},
	1983/2/8.19/{first TOF PET\\BaF$_2$, abandoned},
	1986/1/9.5/{Casey \& Nutt\\block detector},
	2006/1/12.1/{commercial TOF PET\\on LSO/LYSO},
	2018/2/13.4/{uEXPLORER\\total-body}}
{
	\pgfmathsetmacro{\tx}{(\year-1932)*0.155}
	\pgfmathsetmacro{\lead}{0.55 + 1.50*(\r-1)}
	\draw[tlpositron, tick] (\tx,0) -- (\tx,0.25);
	\fill[tlpositron] (\tx,0) circle (1.6pt);
	\draw[tlpositron, lead] (\tx,0.25) -- (\tx,0.40) -- (\lx,0.40) -- (\lx,\lead);
	\begin{pgfonlayer}{tllabels}
	\node[lbl, anchor=south, name=n\year] at (\lx,\lead) {\textbf{\year}\\\text};
	\end{pgfonlayer}
}

% --- below the axis --  year/row/label x/colour/label ----------------------
% Mostly the borrowed lineage (tlsingle); 1951, 1989 and 2013 are positron
% entries below purely for room and keep tlpositron. Say so in the caption.
\foreach \year/\r/\lx/\col/\text in {%
	1951/2/2.3/tlpositron/{Wrenn \emph{et al.}\\coincidence proposed},
	1958/1/3.9/tlsingle/{Anger\\scintillation camera},
	1963/2/5.26/tlsingle/{Kuhl \& Edwards\\emission tomography},
	1973/1/6.6/tlsingle/{Hounsfield\\X-ray CT},
	1989/1/9.3/tlpositron/{LSO\\discovered},
	2013/1/12.2/tlpositron/{SiPM\\digital PET}}
{
	\pgfmathsetmacro{\tx}{(\year-1932)*0.155}
	\pgfmathsetmacro{\lead}{0.55 + 1.50*(\r-1)}
	\draw[\col, tick] (\tx,0) -- (\tx,-0.25);
	\fill[\col] (\tx,0) circle (1.6pt);
	\draw[\col, lead] (\tx,-0.25) -- (\tx,-0.40) -- (\lx,-0.40) -- (\lx,-\lead);
	\begin{pgfonlayer}{tllabels}
	\node[lbl, anchor=north, name=n\year] at (\lx,-\lead) {\textbf{\year}\\\text};
	\end{pgfonlayer}
}
