\newcommand{\topics}[1]{{\small\itshape\color{scaffoldnote}\raggedright
	Touch on: #1\par}}

% Drafting scaffold: live sentence-count badges per heading, read from
% sentence-counts.tex (regenerated each build by tools/count_sentences.py).
\usepackage[explicit]{titlesec}
\IfFileExists{sentence-counts.tex}{\input{sentence-counts.tex}}{}
\newcommand{\sentbadge}[1]{%
	\ifcsname sc@#1\endcsname
		\space{\normalfont\small\color{scaffolddim}[\csname sc@#1\endcsname~sentences]}%
	\fi}

% Drafting scaffold: a needed citation not yet in references.bib. Not a fake
% \cite key -- that would bury biber's genuine missing-citation warnings.
\newcommand{\needcite}[1]{{\normalfont\small\color{scaffoldwarn}[cite: #1]}}

% Drafting scaffold: hand-typed-unit badges, read from unit-warnings.tex
% (regenerated each build by tools/check_units.py). No badge means clean.
\IfFileExists{unit-warnings.tex}{\input{unit-warnings.tex}}{}
\newcommand{\unitbadge}[1]{%
	\ifcsname uw@#1\endcsname
		\space{\normalfont\small\color{scaffoldwarn}%
			[\csname uw@#1\endcsname~raw units]}%
	\fi}

\titleformat{\chapter}[display]

	{\normalfont\Large\bfseries\raggedright}{\thesection}{1em}
	{#1\sentbadge{\thesection}\unitbadge{\thesection}}
\titleformat{\subsection}
	{\normalfont\large\bfseries\raggedright}{\thesubsection}{1em}
	{#1\sentbadge{\thesubsection}\unitbadge{\thesubsection}}
