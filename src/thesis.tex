% Shared by the unit shorthands below. An argument containing a hyphen is a
% range and goes to \qtyrange (\ns{4-12} -> "4 to 12 ns"); anything else is a
% plain \qty. expl3 is part of the LaTeX kernel, so this costs no package.
% ponytail: a leading minus reads as a range, so no \keV{-5}. None of these
% quantities go negative; if one ever does, switch that macro back to \qty.
\ExplSyntaxOn
\NewDocumentCommand \unitvalue { mm }
	{
	\tl_if_in:nnTF {#1} {-}
		{ \unit_range:nn {#1} {#2} }
		{ \qty {#1} {#2} }
	}
\cs_new_protected:Npn \unit_range:nn #1#2
	{ \unit_range_aux:w #1 \q_stop {#2} }
\cs_new_protected:Npn \unit_range_aux:w #1 - #2 \q_stop #3
	{ \qtyrange {#1} {#2} {#3} }
\ExplSyntaxOff
